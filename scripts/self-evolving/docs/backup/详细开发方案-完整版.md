def _dedup_candidates(candidates: list[dict]) -> list[dict]:
    """按 node_id 去重，同 node_id 只保留第一个"""
    seen_nodes: set[str] = set()
    result: list[dict] = []

## 任务 4：C-P1-4 本地重要性缓存 🟡 P1

| **实现方式** | 自己实现 — `defaultdict(int)` 计数器 + `score *= 1 + 0.1 * hit_count`，约 10 行 |
| **批次** | 第二批 |

```python
# ~/.hermes/plugins/knowledge-navigation/src/core/hooks.py
from collections import defaultdict

class HitCounter:
    def __init__(self, decay: float = 0.95):
        self._counts: dict[str, int] = defaultdict(int)
        self._decay = decay

    def boost_score(self, memory: dict) -> float:
        key = (memory.get("text", "") or "")[:50]
        hits = self._counts.get(key, 0)
        base_score = memory.get("rerank_score", 0.5)
        return base_score * (1 + 0.1 * hits)

hit_counter = HitCounter()
# pre_llm_call 中替换 rerank_score:
for m in memories:
    m["rerank_score"] = hit_counter.boost_score(m)
```

# ~/.hermes/plugins/knowledge-navigation/src/core/hooks.py
class TaskTracker:
    def __init__(self, interval: int = 5):
        self._turn_count = 0
        self._interval = interval

    def get_summary_prompt(self, task_goal: str) -> str | None:
        self._turn_count += 1
        if self._turn_count % self._interval != 0:
            return None
        return (
            "[任务状态摘要]\n"
            f"原始任务目标：{task_goal}\n"
            f"当前进度：第 {self._turn_count} 轮\n"
            "请简要总结当前进展并确认是否仍在正向推进。"

tracker = TaskTracker(interval=5)
summary = tracker.get_summary_prompt(task_goal)

# ~/.hermes/plugins/tool_quality/src/core/hooks.py
from collections import defaultdict

class ToolQualityPlugin:
    def __init__(self, max_chars: int = 2000, max_retries: int = 3):
        self._max_chars = max_chars
        self._max_retries = max_retries
        self._failure_count: dict[str, int] = defaultdict(int)

    def on_tool_call(self, tool_name: str, args: dict) -> dict | None:
        if self._failure_count.get(tool_name, 0) >= self._max_retries:
            return {
                "action": "inject_prompt",
                "content": f"[警告] 工具 {tool_name} 已连续失败 {self._max_retries} 次"
            }
        return None

    def on_tool_result(self, tool_name: str, result: str) -> str:
        if len(result) > self._max_chars:
            return result[:self._max_chars] + f"...[truncated, max_chars={self._max_chars}]"
        return result

    def record_failure(self, tool_name: str) -> None:
        self._failure_count[tool_name] += 1

    def record_success(self, tool_name: str) -> None:
        self._failure_count[tool_name] = 0
```

# collect_baseline.py — 新增 bootstrap CI + t-test + 时间窗口
import numpy as np
from scipy.stats import ttest_ind

def bootstrap_ci(values: list[float], n_resamples: int = 1000, ci: float = 0.95) -> dict:
    arr = np.array(values)
    means = [np.mean(np.random.choice(arr, size=len(arr), replace=True)) for _ in range(n_resamples)]
    return {
        "mean": np.mean(means),
        "ci_95_lower": np.percentile(means, (1 - ci) / 2 * 100),
        "ci_95_upper": np.percentile(means, (1 + ci) / 2 * 100)
    }

def ttest_vs_baseline(current: list[float], previous: list[float]) -> dict:
    """Welch's t-test（不等方差）"""
    t_stat, p = ttest_ind(current, previous, equal_var=False)
    return {"t_statistic": float(t_stat), "p_value": float(p)}
```

) -> tuple[np.ndarray, np.ndarray]:
    hdb = HDBSCAN(
        min_cluster_size=min_cluster_size,
        cluster_selection_method=cluster_selection_method,
        metric="euclidean",
    )
    labels = hdb.fit_predict(embeddings)
    probabilities = hdb.probabilities_
    n_clusters = len(set(labels) - {-1})
    n_noise = (labels == -1).sum()
    print(f"HDBSCAN: {n_clusters} 簇, {n_noise} 噪声")
    return labels, probabilities

# cli.py 改动：Rounds 2-4 的三次循环 → 单次 HDBSCAN 调用

# prefilter.py
from datetime import datetime, timezone

def forgetting_curve_score(
    mentioned_at: str | None, created_at: str | None,
    half_life_days: float = 7.0
) -> float:
    now = datetime.now(timezone.utc)
    ref_time = None
    for t in [mentioned_at, created_at]:
        if t:
            try:
                parsed = datetime.fromisoformat(t.replace("Z", "+00:00"))
                if ref_time is None or parsed > ref_time:
                    ref_time = parsed
            except (ValueError, TypeError):
                pass
    if ref_time is None:
        return 1.0
    elapsed_days = (now - ref_time).total_seconds() / 86400
    return 0.5 ** (elapsed_days / half_life_days)

def apply_prefilter(entries: list[dict]) -> tuple[list[dict], list[int]]:
    keep, remove_indices = [], []
    for i, e in enumerate(entries):
        score = forgetting_curve_score(e.get("mentioned_at"), e.get("created_at"))
        if score < 0.05:
            remove_indices.append(i)
        else:
            keep.append(e)
    return keep, remove_indices

# compressor.py
def semantic_merge(entries: list[str], llm_client) -> str:
    prompt = f"""
    以下是 {len(entries)} 条相关记忆，请合并为 1 条精炼摘要：
    {'---'.join(entries)}
    要求：保留所有不重复的关键信息，去除冗余/过时内容，输出 1-2 句精炼摘要
    """
    return llm_client.call(prompt)
